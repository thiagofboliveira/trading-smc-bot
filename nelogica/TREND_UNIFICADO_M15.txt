// ==============================================================================
// ESTRATÉGIA: SUPER TENDÊNCIA UNIFICADA M15 (MULTI-GATILHO SINGLE-POSITION)
// Plataforma: Nelogica ProfitChart / Rico Trader / Clear Trader / BlackArrow
// Linguagem: NTSL (Nelogica Trading System Language)
// Versão: 1.1 (Revisada pós-auditoria de simulação)
// ==============================================================================
// ⚠️ AVISO CRÍTICO DE AUDITORIA E DIVERGÊNCIAS (LEIA ANTES DE USAR):
// 1. NÃO ALINHADO INTEGRALMENTE COM O PYTHON:
//    - O ProfitChart NTSL não possui ordens OCO intrabar nativas via script.
//      Ao acionar a entrada por BuyStop, o SellToCoverStop só é registrado no
//      fechamento da barra, deixando a barra de entrada vulnerável no backtest do Profit.
//      Para execução real, configure uma regra OCO na Boleta/Roteamento da corretora!
// 2. ATIVOS DA B3 (WIN / WDO) NÃO TESTADOS:
//    - Este modelo foi pesquisado em NAS100 e US30. Testes em Forex e Commodities
//      falharam (-31R a -63R). NUNCA opere no mini-índice sem validação prévia.
// 3. RESULTADO HONESTO DO MODELO CONFIGURADO:
//    - No Python (modo single-position com 1 pt de slippage), o resultado em 5 anos
//      é de apenas +54.7R em 457 trades (R/tr: +0.120 ± 0.112, t = 1.07, PF 1.18).
//      Estatisticamente não possui vantagem comprovada sobre o zero. NÃO USAR EM REAL.
// ==============================================================================

Parametro
  // Seleção de Gatilhos (Fila de Prioridade: Trap > Força > Engolfo)
  UsarTrap(Verdadeiro);             // Ativa gatilho de Trap na MME 8
  UsarBarraForca(Verdadeiro);       // Ativa gatilho de Barra de Força / Ponto Contínuo
  UsarEngolfo(Verdadeiro);          // Ativa gatilho de Engolfo na Média
  
  // Médias Móveis e Indicadores
  PeriodoMME_Conducao(20);         // MME 20 (Suporte Dinâmico Institucional)
  PeriodoMME_Fast(80);             // MME 80 (Tendência Primária)
  PeriodoMME_Slow(200);            // MME 200 (Macro Tendência)
  PeriodoMME_Gatilho(8);           // MME 8 (Média de Timing do Trap)
  PeriodoADX(14);                  // ADX para filtro de super tendência
  MinADX(25.0);                    // ADX mínimo exigido
  MinSlope(0.10);                  // Inclinação mínima da MME 80 normalizada por ATR
  MaxSlope(1.00);                  // Inclinação máxima da MME 80 normalizada por ATR

  // Gerenciamento de Risco Diário
  MaxTradesDia(3);                 // Máximo de operações por dia
  MaxStopsDia(2);                  // Trava de risco diário após 2 stops
  PermitirOvernight(Falso);        // Python mantém posições overnight; no Profit padrão é day-trade

  // Horários Operacionais (Horário de Brasília)
  HorarioInicioEntrada(1000);      // 10:00 BRT (13:00 UTC - Abertura de NY)
  HorarioLimiteEntrada(1430);      // 14:30 BRT (17:30 UTC - Fim do Miolo de NY)
  HorarioZeragem(1745);            // Zeragem obrigatória se day-trade

Var
  mme8                  : Real;
  mme20                 : Real;
  mme80                 : Real;
  mme200                : Real;
  adxVal                : Real;
  atrVal                : Real;
  slope80               : Real;
  cRange                : Real;
  
  // Condições Macro (medidas em i-1)
  tendenciaAlta         : Booleano;
  tendenciaBaixa        : Booleano;
  adxForte              : Booleano;
  slopeBullish          : Booleano;
  slopeBearish          : Booleano;
  emHorario             : Booleano;

  // Sinais de Gatilho
  sinalTrapCompra       : Booleano;
  sinalTrapVenda        : Booleano;
  sinalForcaCompra      : Booleano;
  sinalForcaVenda       : Booleano;
  sinalEngolfoCompra    : Booleano;
  sinalEngolfoVenda     : Booleano;

  // Preços de Execução
  precoGatilho          : Real;
  precoStopLoss         : Real;

  // Controle Diário
  tradesHoje            : Inteiro;
  stopsHoje             : Inteiro;
  diaAtual              : Inteiro;
  posicaoAnterior       : Inteiro;

Inicio
  // 1. Inicialização do Dia
  Se (Date <> diaAtual) entao
  Inicio
    diaAtual := Date;
    tradesHoje := 0;
    stopsHoje := 0;
    posicaoAnterior := 0;
  Fim;

  // 2. Rastreamento de Trades Executados
  Se (HasPosition) e (posicaoAnterior = 0) entao
  Inicio
    tradesHoje := tradesHoje + 1;
    posicaoAnterior := 1;
  Fim
  Senao Se (Not HasPosition) e (posicaoAnterior = 1) entao
  Inicio
    posicaoAnterior := 0;
  Fim;

  // 3. Cálculo dos Indicadores Técnicos
  mme8 := MediaExp(PeriodoMME_Gatilho, Fechamento);
  mme20 := MediaExp(PeriodoMME_Conducao, Fechamento);
  mme80 := MediaExp(PeriodoMME_Fast, Fechamento);
  mme200 := MediaExp(PeriodoMME_Slow, Fechamento);
  adxVal := ADX(PeriodoADX, Fechamento);
  atrVal := AvgTrueRange(14);
  cRange := Maxima - Minima;

  // Inclinação normalizada da MME 80 em 10 barras por ATR(14) (idêntico ao Python)
  Se (atrVal[1] > 0) entao
    slope80 := (mme80[1] - mme80[11]) / (10 * atrVal[1])
  Senao
    slope80 := 0;

  // 4. Filtros Macro Causais em i-1
  tendenciaAlta := (Fechamento[1] > mme80[1]) e (mme80[1] > mme200[1]);
  tendenciaBaixa := (Fechamento[1] < mme80[1]) e (mme80[1] < mme200[1]);
  adxForte := (adxVal[1] >= MinADX);
  slopeBullish := (slope80 >= MinSlope) e (slope80 <= MaxSlope);
  slopeBearish := (slope80 <= -MinSlope) e (slope80 >= -MaxSlope);
  emHorario := (Time >= HorarioInicioEntrada) e (Time <= HorarioLimiteEntrada);

  // 5. Gerenciamento de Posição Aberta
  Se (IsBought) entao
  Inicio
    // Stop Loss Fixo da Posição
    SellToCoverStop(precoStopLoss, precoStopLoss);

    // Condução Causal pela MME 20: encerra no fechamento abaixo da média
    Se (Fechamento < mme20) entao
    Inicio
      ClosePosition;
    Fim;
  Fim
  Senao Se (IsSold) entao
  Inicio
    // Stop Loss Fixo da Posição
    BuyToCoverStop(precoStopLoss, precoStopLoss);

    // Condução Causal pela MME 20: encerra no fechamento acima da média
    Se (Fechamento > mme20) entao
    Inicio
      ClosePosition;
    Fim;
  Fim;

  // Zeragem de fim de dia (apenas se day-trade)
  Se (Not PermitirOvernight) e (Time >= HorarioZeragem) e (HasPosition) entao
  Inicio
    ClosePosition;
  Fim;

  // 6. Detecção e Arme Imediato de Gatilhos (Válidos para a Barra Seguinte i+1)
  Se (Not HasPosition) e (emHorario) e (adxForte) e (cRange > 0) e (tradesHoje < MaxTradesDia) e (stopsHoje < MaxStopsDia) entao
  Inicio
    // --- COMPRA (Super Tendência de Alta com Slope Válido) ---
    Se (tendenciaAlta) e (slopeBullish) entao
    Inicio
      // GATILHO 1: TRAP MME 8 (Prioridade 1)
      sinalTrapCompra := UsarTrap e (Minima <= mme8 * 1.001) e (Minima < Minima[1]) e (Fechamento > Minima[1]) e (Fechamento > Abertura);
      
      // GATILHO 2: BARRA DE FORÇA (Prioridade 2)
      sinalForcaCompra := UsarBarraForca e (Minima <= Max(mme8, mme20) * 1.001) e (Fechamento > Abertura) e ((Fechamento - Minima) / cRange >= 0.70) e (Fechamento > mme8);
      
      // GATILHO 3: ENGOLFO (Prioridade 3)
      sinalEngolfoCompra := UsarEngolfo e (Minima <= Max(mme8, mme20) * 1.001) e (Fechamento[1] < Abertura[1]) e (Fechamento > Abertura) e (Fechamento >= Abertura[1]) e (Abertura <= Fechamento[1]);

      Se (sinalTrapCompra) entao
      Inicio
        PaintBar(clYellow);
        precoGatilho := Maxima;
        precoStopLoss := Minima;
        BuyStop(precoGatilho, precoGatilho);
      Fim
      Senao Se (sinalForcaCompra) entao
      Inicio
        PaintBar(clBlue);
        precoGatilho := Maxima;
        precoStopLoss := Minima;
        BuyStop(precoGatilho, precoGatilho);
      Fim
      Senao Se (sinalEngolfoCompra) entao
      Inicio
        PaintBar(clAqua);
        precoGatilho := Maxima;
        precoStopLoss := Minima;
        BuyStop(precoGatilho, precoGatilho);
      Fim;
    Fim

    // --- VENDA (Super Tendência de Baixa com Slope Válido) ---
    Senao Se (tendenciaBaixa) e (slopeBearish) entao
    Inicio
      // GATILHO 1: TRAP MME 8
      sinalTrapVenda := UsarTrap e (Maxima >= mme8 * 0.999) e (Maxima > Maxima[1]) e (Fechamento < Maxima[1]) e (Fechamento < Abertura);
      
      // GATILHO 2: BARRA DE FORÇA
      sinalForcaVenda := UsarBarraForca e (Maxima >= Min(mme8, mme20) * 0.999) e (Fechamento < Abertura) e ((Maxima - Fechamento) / cRange >= 0.70) e (Fechamento < mme8);
      
      // GATILHO 3: ENGOLFO
      sinalEngolfoVenda := UsarEngolfo e (Maxima >= Min(mme8, mme20) * 0.999) e (Fechamento[1] > Abertura[1]) e (Fechamento < Abertura) e (Fechamento <= Abertura[1]) e (Abertura >= Fechamento[1]);

      Se (sinalTrapVenda) entao
      Inicio
        PaintBar(clYellow);
        precoGatilho := Minima;
        precoStopLoss := Maxima;
        SellShortStop(precoGatilho, precoGatilho);
      Fim
      Senao Se (sinalForcaVenda) entao
      Inicio
        PaintBar(clBlue);
        precoGatilho := Minima;
        precoStopLoss := Maxima;
        SellShortStop(precoGatilho, precoGatilho);
      Fim
      Senao Se (sinalEngolfoVenda) entao
      Inicio
        PaintBar(clAqua);
        precoGatilho := Minima;
        precoStopLoss := Maxima;
        SellShortStop(precoGatilho, precoGatilho);
      Fim;
    Fim;
  Fim;

Fim;
