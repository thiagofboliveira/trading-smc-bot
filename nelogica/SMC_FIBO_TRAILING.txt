// ==============================================================================
// PLAYBOOK OFICIAL SMC-FIBO-TRAILING UNIVERSAL (NTSL / PROFIT / RICO TRADER)
// ==============================================================================
// Estratégia: SMC Fibonacci OTE (Optimal Trade Entry) + Condução Livre MME 8
// Versão: 2.0 UNIVERSAL (Auto-Adaptação Dinâmica de Volatilidade para QUALQUER ATIVO)
// Compatível com: Mini-Índice (WIN), Mini-Dólar (WDO), Ações (PETR4, VALE3), BDRs, Cripto
// ==============================================================================

Parametro
  FiboEntrada(0.705);       // Nível Sweet Spot Institucional (70.5%)
  FiboStop(0.790);          // Nível de Invalidação Curto (79.0%)
  PeriodoMME(8);            // Média Móvel Exponencial para Condução do Trade
  MultVolatilidade(1.5);    // Multiplicador do Range Médio (Auto-escala para qualquer ativo)
  PeriodoVolatilidade(20);  // Barras para calcular o tamanho médio do candle

Var
  mme8              : Real;
  rangeMedio        : Real;
  filtroMinimo      : Real;
  topoFractal       : Booleano;
  fundoFractal      : Booleano;
  ultimoTopo        : Real;
  ultimoFundo       : Real;
  
  emOperacao        : Inteiro;  // 0 = Fora, 1 = Comprado, -1 = Vendido
  precoEntrada      : Real;
  precoStop         : Real;
  precoAlvo         : Real;
  riscoInicial      : Real;
  
  pernadaAltaAtiva  : Booleano;
  pernadaBaixaAtiva : Booleano;
  refTopo           : Real;
  refFundo          : Real;
  diffPernada       : Real;
  nivelEntrada      : Real;
  nivelStop         : Real;

Inicio
  // 1. Cálculo da MME de Condução
  mme8 := MediaExp(PeriodoMME, Fechamento);

  // 2. Cálculo Universal de Volatilidade (Auto-Escala Dinâmica)
  // Em vez de pontos fixos que só funcionam no WIN, calcula o tamanho médio dos candles do ativo
  rangeMedio   := Media(PeriodoVolatilidade, Maxima - Minima);
  filtroMinimo := rangeMedio * MultVolatilidade;

  // 3. Detecção de Swings Fractais (Atraso de 2 barras para confirmação sem repintura)
  topoFractal := (Maxima[2] > Maxima[1]) e (Maxima[2] > Maxima[0]) e 
                 (Maxima[2] > Maxima[3]) e (Maxima[2] > Maxima[4]);
                 
  fundoFractal := (Minima[2] < Minima[1]) e (Minima[2] < Minima[0]) e 
                  (Minima[2] < Minima[3]) e (Minima[2] < Minima[4]);

  Se topoFractal Entao
    ultimoTopo := Maxima[2];

  Se fundoFractal Entao
    ultimoFundo := Minima[2];

  // 4. Detecção de Quebra de Estrutura (BOS - Break of Structure)
  Se (Not HasPosition) e (emOperacao = 0) Entao
  Inicio
    // BOS de Alta: Fechamento rompe o último topo institucional com perna relevante
    Se (ultimoTopo > 0) e (ultimoFundo > 0) e (Fechamento > ultimoTopo) e (Fechamento[1] <= ultimoTopo) Entao
    Inicio
      diffPernada := Maxima - ultimoFundo;
      Se diffPernada >= filtroMinimo Entao
      Inicio
        pernadaAltaAtiva  := Verdadeiro;
        pernadaBaixaAtiva := Falso;
        refTopo           := Maxima;
        refFundo          := ultimoFundo;
      Fim;
    Fim;

    // BOS de Baixa: Fechamento perde o último fundo institucional com perna relevante
    Se (ultimoTopo > 0) e (ultimoFundo > 0) e (Fechamento < ultimoFundo) e (Fechamento[1] >= ultimoFundo) Entao
    Inicio
      diffPernada := ultimoTopo - Minima;
      Se diffPernada >= filtroMinimo Entao
      Inicio
        pernadaBaixaAtiva := Verdadeiro;
        pernadaAltaAtiva  := Falso;
        refTopo           := ultimoTopo;
        refFundo          := Minima;
      Fim;
    Fim;
  Fim;

  // Atualizar expansão da pernada enquanto não houver retração profunda
  Se pernadaAltaAtiva e (Maxima > refTopo) Entao
    refTopo := Maxima;

  Se pernadaBaixaAtiva e (Minima < refFundo) Entao
    refFundo := Minima;

  // 5. Execução de Entrada no Sweet Spot 70.5% (Se não estiver posicionado)
  Se (Not HasPosition) e (emOperacao = 0) Entao
  Inicio
    // Entrada Comprada
    Se pernadaAltaAtiva Entao
    Inicio
      diffPernada  := refTopo - refFundo;
      nivelEntrada := refTopo - (diffPernada * FiboEntrada);
      nivelStop    := refTopo - (diffPernada * FiboStop);

      Se (Minima <= nivelEntrada) e (Fechamento > nivelStop) Entao
      Inicio
        BuyAtMarket;
        emOperacao        := 1;
        precoEntrada      := nivelEntrada;
        precoStop         := nivelStop;
        precoAlvo         := refTopo;
        riscoInicial      := precoEntrada - precoStop;
        pernadaAltaAtiva  := Falso;
      Fim;
    Fim;

    // Entrada Vendida
    Se pernadaBaixaAtiva Entao
    Inicio
      diffPernada  := refTopo - refFundo;
      nivelEntrada := refFundo + (diffPernada * FiboEntrada);
      nivelStop    := refFundo + (diffPernada * FiboStop);

      Se (Maxima >= nivelEntrada) e (Fechamento < nivelStop) Entao
      Inicio
        SellShortAtMarket;
        emOperacao        := -1;
        precoEntrada      := nivelEntrada;
        precoStop         := nivelStop;
        precoAlvo         := refFundo;
        riscoInicial      := precoStop - precoEntrada;
        pernadaBaixaAtiva := Falso;
      Fim;
    Fim;
  Fim;

  // 6. Gestão Ativa da Posição (Condução 100% Livre pela MME 8 até Inversão ou Alvo)
  Se HasPosition ou (emOperacao <> 0) Entao
  Inicio
    // -------------------------------------------------------------
    // GESTÃO DA COMPRA (Trailing Livre)
    // -------------------------------------------------------------
    Se (IsBought) ou (emOperacao = 1) Entao
    Inicio
      PaintBar(clVerde); // Coloração verde durante a condução

      // A) STOP LOSS ORIGINAL (Proteção se o trade não evoluir)
      Se (Minima <= precoStop) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
      Fim

      // B) SAÍDA DINÂMICA POR TRAILING STOP NA MME 8
      // Encerra imediatamente quando uma barra fecha abaixo da MME 8 com lucro
      Senao Se (Fechamento < mme8) e (Fechamento[1] >= mme8[1]) e (Fechamento > precoEntrada) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
        Alert(clAzul); // Alerta visual/sonoro: Saída Lucrativa por Trailing MME 8!
      Fim

      // C) SAÍDA NO TOPO PRÉVIO ESTRUTURAL (Se atingir a máxima absoluta)
      Senao Se (Maxima >= precoAlvo) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
      Fim;
    Fim;

    // -------------------------------------------------------------
    // GESTÃO DA VENDA (Trailing Livre)
    // -------------------------------------------------------------
    Se (IsSold) ou (emOperacao = -1) Entao
    Inicio
      PaintBar(clVermelho); // Coloração vermelha durante a condução

      // A) STOP LOSS ORIGINAL (Proteção se o trade não evoluir)
      Se (Maxima >= precoStop) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
      Fim

      // B) SAÍDA DINÂMICA POR TRAILING STOP NA MME 8
      // Encerra imediatamente quando uma barra fecha acima da MME 8 com lucro
      Senao Se (Fechamento > mme8) e (Fechamento[1] <= mme8[1]) e (Fechamento < precoEntrada) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
        Alert(clAzul); // Alerta visual/sonoro: Saída Lucrativa por Trailing MME 8!
      Fim

      // C) SAÍDA NO FUNDO PRÉVIO ESTRUTURAL (Se atingir a mínima absoluta)
      Senao Se (Minima <= precoAlvo) Entao
      Inicio
        ClosePosition;
        emOperacao := 0;
      Fim;
    Fim;
  Fim;

Fim;
